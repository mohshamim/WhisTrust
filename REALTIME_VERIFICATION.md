```typescript
// Realtime message handler with duplicate prevention
const handleNewMessage = useCallback((message: Message) => {
  setCaseData((prev) => {
    if (!prev) return prev;
    
    // Prevent duplicates
    const messageExists = prev.messages.some((m) => m.id === message.id);
    if (messageExists) return prev;

    const newMessages = [
      ...prev.messages,
      {
        id: message.id,
        type: message.sender_type,
        text: message.content,
        timestamp: new Date(message.created_at).toLocaleString(),
      },
    ];

    // Auto-scroll after state update
    setTimeout(() => {
      if (messagesEndRef.current) {
        messagesEndRef.current.scrollIntoView({ behavior: "smooth" });
      }
    }, 50);

    return { ...prev, messages: newMessages };
  });
}, []);
```

### **4. Database Schema** ✅ VERIFIED

**Location**: `supabase/schema.sql`

**Messages Table Structure**:
```sql
CREATE TABLE IF NOT EXISTS messages (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  report_id UUID NOT NULL REFERENCES reports(id) ON DELETE CASCADE,
  sender_type TEXT NOT NULL CHECK (sender_type IN ('reporter', 'admin')),
  sender_id UUID REFERENCES auth.users(id) ON DELETE SET NULL,
  content TEXT NOT NULL,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
```

**Verified Constraints**:
- ✅ Primary key on `id` (UUID, auto-generated)
- ✅ Foreign key to `reports.id` with CASCADE delete
- ✅ CHECK constraint on `sender_type` ('reporter' or 'admin')
- ✅ Auto-generated timestamp on `created_at`
- ✅ Indexes on `report_id` and `created_at` for performance

**Indexes**:
```sql
CREATE INDEX idx_messages_report_id ON messages(report_id);
CREATE INDEX idx_messages_created_at ON messages(created_at);
```
